# oppo-dv-mkv: options
#
# Change a value, keep this file next to the AutoScript folder on the USB stick and run the installer again. Running
# the installer without this file keeps the options already installed. "stock" (or no value) = the player's own way.
# The same options can be changed on the player: Setup > Extras (the page writes this file on the player; a copy on
# the stick replaces them at the next install).
#
# dv_mkv = on | off
#   Dolby Vision from MKV files (profile 5 and 8 single layer, profile 7 with BL and EL in one track). off = the
#   player's own way: such files play as HDR10 / SDR (profile 5 has no HDR10 base: wrong colours). Everything else
#   (MP4 subtitles and languages, the profile 7 and TV-led metadata fixes) stays. From Setup > Extras: at once.
dv_mkv = on
#
# dv_l5 = on | off
#   TV-led Dolby Vision: the active area (L5 metadata) goes to the TV, rewritten for the frame the player outputs and
#   widened for what it draws in the bars. off = Oppo's way: no L5 (the TV then also ignores positive lift); the other
#   metadata fixes stay. From Setup > Extras: at once.
dv_l5 = on
#
# forced_subs = on | off
#   MKV files: start on the subtitle track with the Forced flag (MKVToolNix "Forced display") in the language set in
#   Setup > Playback Setup > Language > Subtitle Language (Auto: the player's menu language), for the lines in another
#   language; no such track = subtitles off. off = the player's own choice (with Subtitle Language = Auto: the track in
#   the menu language, see the README). Subtitles chosen during the film are not changed. From Setup > Extras: at once.
forced_subs = off
#
# native_25p = on | off
#   HD video (MKV and MP4, H.264 or HEVC, 720 lines and up) at 25, 29.97 or 30 frames per second plays at that rate:
#   Output Resolution Auto -> 2160p25 / 2160p30 (when the TV accepts it), Source Direct -> 1080p25 / 1080p30.
#   off = stock: it plays at 50 / 60 Hz (every frame shown twice; the motion looks the same), because the player is
#   never told that such a file is progressive and takes it for interlaced.
#   Interlaced video (1080i TV recordings, concert and TV-series discs) is recognised and plays as before, at 50 / 60 Hz.
#   Not changed: SD files, 720p in Source Direct (HDMI has no 720p25), TS / M2TS files, MPEG-2 and VC-1 video.
native_25p = off
#
# menu_refresh = 60 | 50 | stock
#   Refresh rate of the home menu and screen saver (nothing playing) with Output Resolution Auto, UHD Auto, 1080p Auto
#   or Source Direct. stock: RU, EU and AU firmware use 50 Hz (and stay at 25 Hz after a 4K 25p file).
#   US, JP, CN and TW firmware already use 60 Hz; there this option does nothing.
menu_refresh = stock
#
# default_language = en | de | fr | es | it | sv | nl | pl | zh | zh-tw | ja | ru | stock
#   The language that Setup > Device Setup > Reset Factory Defaults sets for the player's menus, disc menus and
#   audio (subtitles: Auto, or the same language for zh, ja, ru). Your current settings are not changed.
#   zh = Simplified Chinese menus. zh-tw = Traditional Chinese menus, English disc menus and audio, Chinese
#   subtitles (what the TW firmware sets).
#   stock: the firmware's own (Russian on RU firmware, English on US/EU/AU).
default_language = stock
#
# boot_message = on | off
#   After every boot the player shows for 10 seconds which oppo-dv-mkv patches loaded, in its own popup:
#   "oppo-dv-mkv <version>: all patches loaded". off = no message when everything loaded. If a part did NOT load,
#   the message names it, also with off ("oppo-dv-mkv: NOT loaded: ..."). After a boot on which the fail-safe skipped
#   the patches it says so. RU 65-0131 player program and its jailbreak copies only (RU and USA jailbreak, jb_overlay).
boot_message = on
#
# dlna_iso = on | off
#   Blu-ray, UHD and DVD disc images (ISO) and disc folders (BDMV / VIDEO_TS on a WebDAV server) play over DLNA:
#   pushed from a phone or tablet app, or picked in Network > DLNA server. Needs the Dolby Vision MKV fix and the
#   jailbreak's ISO player (RU and USA jailbreak, jb_overlay). off = the player's own DLNA playback only.
dlna_iso = on
#
# setup_extras = on | off
#   The "Extras" page in the player's Setup menu (between Network Setup and Exit) to change these options on the
#   player. off = Oppo's Setup menu as it is. RU 65-0131 player program and its jailbreak copies only.
setup_extras = on
